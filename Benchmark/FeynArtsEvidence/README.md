Run it only with host headroom and a fresh scope, for example from `Benchmark/`:

```bash
unit="feyngrav-1pi-audit-$(date +%s)-$$"
systemd-run --user --scope --same-dir --unit="$unit" \
  -p MemoryAccounting=yes -p MemoryMax=1G -p MemorySwapMax=0 \
  -p OOMPolicy=kill -p KillMode=control-group \
  -p RuntimeMaxSec=175s -p TimeoutStopSec=7s -p TasksMax=48 \
  -- /usr/bin/timeout --kill-after=5s 155s /usr/local/bin/WolframKernel \
  -noinit -script FeynArtsEvidence/check-1pi-source-factors-cgroup.wls \
  > FeynArtsEvidence/check-1pi-source-factors-cgroup.log 2>&1
```
